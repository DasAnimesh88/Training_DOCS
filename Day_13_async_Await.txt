function getData(dataId, nextData) {
    setTimeout(() => {
        console.log("data", dataId);
        if (nextData) {
            nextData();
        }
    }, 2000);
}

Promise 

function getData(dataId) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            console.log("data", dataId);
            resolve("success");
        }, 2000);
    });
}

// Promise chaining
getData(1)
    .then(() => getData(2))
    .then(() => getData(3))
    .then(() => {
        console.log("done");
    })
    .catch((err) => {
        console.log(err);
    });

What is IIFE?

IIFE stands for Immediately Invoked Function Expression. 
It's a function that runs immediately after it's defined, without needing to call it separately.
