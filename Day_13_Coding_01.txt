function reverseNumber(num) {
    return Number(
        String(num).split("").reverse().join("")
    );
}

console.log(reverseNumber(12345));  // Output: 54321
console.log(reverseNumber(1200));   // Output: 21 (trailing zeros removed)

A palindrome is a string that reads the same forwards and backwards.

Examples

"madam"    → true
"racecar"  → true
"hello"    → false

function isPallindrome(str){
    let newWord=str.split("").reverse().join("");

        console.log("it is pallindrome")
    }
    else{
        console.log("it is not pallindrome")
    }


}

function isPalindrome(str) {
    return str.toLowerCase() === str.split("").reverse().join("").toLowerCase();
}

Two strings are anagrams if they contain the same characters with the same frequency, 
regardless of order.

"listen"  ↔ "ilents"   → true
"race"    ↔ "care"     → true

"raec"    ↔ "arce"     → true
"hello"   ↔ "world"    → false

function removeDuplicates(str) {
    return str
        .split("")
        .filter((char, index, arr) => arr.indexOf(char) === index)
        .join("");
}

console.log(removeDuplicates("hello")); // "helo"
